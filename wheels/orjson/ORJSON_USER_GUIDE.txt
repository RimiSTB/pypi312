--------------------------------------------------------------------------------
1) WHAT IS ORJSON?
--------------------------------------------------------------------------------

orjson is a fast, correct JSON library written in Rust via PyO3. It provides:
  - 10x faster than stdlib json for serialization
  - 2x faster than ujson
  - Native support for datetime, numpy, dataclasses, and more
  - Strict UTF-8 decoding (no surrogatepass by default)
  - Returns bytes (not str) from dumps()

    import orjson
    print(orjson.__version__)        # 3.11.9
    print(orjson.dumps({"a": 1}))    # b'{"a":1}'

3) BASIC USAGE
--------------------------------------------------------------------------------

    import orjson

    # dumps: dict -> bytes
    b = orjson.dumps({"name": "Alice", "age": 30})
    print(type(b))       # <class 'bytes'>
    print(b)             # b'{"name":"Alice","age":30}'

    # loads: bytes/str -> dict
    data = orjson.loads(b)
    print(data)          # {'name': 'Alice', 'age': 30}

    # loads accepts both bytes and str
    orjson.loads('{"x": 1}')
    orjson.loads(b'{"x": 1}')

4) SUPPORTED DATA TYPES
--------------------------------------------------------------------------------

    import orjson
    from datetime import datetime, timezone, date, time

    # Primitives
    orjson.dumps(42)                # b'42'
    orjson.dumps(3.14)              # b'3.14'
    orjson.dumps("hello")           # b'"hello"'
    orjson.dumps(True)              # b'true'
    orjson.dumps(None)              # b'null'

    # Collections
    orjson.dumps([1, 2, 3])         # b'[1,2,3]'
    orjson.dumps({"a": 1})          # b'{"a":1}'

    # datetime objects (serialized to ISO 8601 strings)
    dt = datetime(2024, 1, 15, 12, 30, tzinfo=timezone.utc)
    orjson.dumps({"ts": dt})        # b'{"ts":"2024-01-15T12:30:00+00:00"}'
    orjson.dumps(date(2024, 1, 1))  # b'"2024-01-01"'
    orjson.dumps(time(12, 30))      # b'"12:30:00"'

    # bytes -> base64 encoded
    orjson.dumps(b"hello")          # b'"aGVsbG8="'

    # numpy arrays (with option)
    import numpy as np
    orjson.dumps(np.array([1, 2, 3]), option=orjson.OPT_SERIALIZE_NUMPY)

    # dataclasses
    from dataclasses import dataclass
    @dataclass
    class User:
        name: str
        age: int
    orjson.dumps(User("Alice", 30))  # b'{"name":"Alice","age":30}'

    # Enums
    from enum import Enum
    class Color(Enum):
        RED = 1
        GREEN = 2
    orjson.dumps(Color.RED)          # b'1'

5) OPTIONS (flags)
--------------------------------------------------------------------------------

    import orjson

    data = {"z": 1, "a": 2, "m": 3}

    # Pretty-print with 2-space indent
    orjson.dumps(data, option=orjson.OPT_INDENT_2)
    # b'{\n  "a": 2,\n  "m": 3,\n  "z": 1\n}'

    # Sort keys alphabetically
    orjson.dumps(data, option=orjson.OPT_SORT_KEYS)
    # b'{"a":2,"m":3,"z":1}'

    # Serialize numpy arrays
    import numpy as np
    orjson.dumps({"arr": np.array([1, 2, 3])},
                 option=orjson.OPT_SERIALIZE_NUMPY)

    # Combine options with |
    orjson.dumps(data, option=orjson.OPT_INDENT_2 | orjson.OPT_SORT_KEYS)

Full list of options:
  OPT_INDENT_2          - 2-space indentation
  OPT_INDENT_4          - 4-space indentation
  OPT_SORT_KEYS         - sort object keys
  OPT_SERIALIZE_NUMPY   - serialize numpy arrays
  OPT_SERIALIZE_DATACLASS - serialize dataclasses (default on)
  OPT_SERIALIZE_ENUM    - serialize enums as values (not names)
  OPT_APPEND_NEWLINE    - append \n to output
  OPT_NON_STR_KEYS      - allow non-string dict keys

7) DATETIME HANDLING

    import orjson
    from datetime import datetime, timezone

    # Timestamps are serialized to ISO 8601
    dt = datetime(2024, 6, 15, 12, 30, 45, 123456, tzinfo=timezone.utc)
    b = orjson.dumps({"ts": dt})
    print(orjson.loads(b))
    # {'ts': '2024-06-15T12:30:45.123456+00:00'}

    # To serialize as epoch seconds instead:
    orjson.dumps({"ts": dt}, option=orjson.OPT_SERIALIZE_DATACLASS)

    # Parse timestamps back to datetime with default strategy
    # orjson.loads returns strings for timestamps, not datetime objects.
    # Use a custom decoder if you need datetime objects.

8) NUMPY SUPPORT
--------------------------------------------------------------------------------

    import orjson
    import numpy as np

    # Enable numpy serialization with the option flag
    arr = np.array([1.0, 2.0, 3.0])
    b = orjson.dumps({"data": arr}, option=orjson.OPT_SERIALIZE_NUMPY)
    result = orjson.loads(b)
    print(result["data"])    # [1.0, 2.0, 3.0]

    # 2D arrays become nested lists
    mat = np.array([[1, 2], [3, 4]])
    b = orjson.dumps(mat, option=orjson.OPT_SERIALIZE_NUMPY)
    print(orjson.loads(b))   # [[1, 2], [3, 4]]

9) ERROR HANDLING
--------------------------------------------------------------------------------

    import orjson

    # Invalid JSON raises JSONDecodeError
    try:
        orjson.loads(b"not valid json")
    except orjson.JSONDecodeError as e:
        print(e)           # "unexpected character at line 1 column 1"

    # Unsupported types raise TypeError
    try:
        orjson.dumps(object())
    except TypeError as e:
        print(e)           # "Type is not JSON serializable: object"

    # Circular references raise ValueError
    try:
        a = {}
        a["self"] = a
        orjson.dumps(a)
    except ValueError as e:
        print(e)           # "Circular reference detected"

11) ORJSON vs JSON (stdlib)
--------------------------------------------------------------------------------

    import json
    import orjson

    data = {"users": [{"id": i, "name": "u%d" % i} for i in range(1000)]}

    # stdlib json
    s = json.dumps(data)           # returns str

    # orjson
    b = orjson.dumps(data)         # returns bytes

    # Key differences:
    # 1. orjson returns bytes, json returns str
    # 2. orjson is 10-100x faster
    # 3. orjson handles datetime, numpy, dataclasses natively
    # 4. orjson does not have .dumps default=str fallback
    # 5. orjson is strict UTF-8 (no surrogatepass by default)


--------------------------------------------------------------------------------
12) PITFALLS & NOTES
--------------------------------------------------------------------------------

- orjson.dumps returns bytes, not str. Use b.decode("utf-8") if you need str.
- orjson is strict about UTF-8. Invalid UTF-8 in strings will raise errors.
- Numpy support requires OPT_SERIALIZE_NUMPY flag. Without it, numpy arrays
  raise TypeError.
- Dataclass serialization is enabled by default. Use OPT_SERIALIZE_DATACLASS
  only if you need to disable it.
- Circular references raise ValueError, not an infinite loop.
- orjson does not have a streaming/incremental API. You must encode/decode
  the entire structure at once.
- orjson handles datetime as ISO 8601 strings. It does not automatically
  parse ISO strings back to datetime objects.
- Wheels are tagged cp312-cp312-linux_<ABI>. Install the correct ABI wheel.
- Built with Rust/PyO3 via maturin. The .so file is a Rust staticlib linked
  into a Python extension.
- No external dependencies. orjson is self-contained.
