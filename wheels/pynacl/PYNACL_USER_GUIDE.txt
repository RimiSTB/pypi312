--------------------------------------------------------------------------------
1) WHAT IS PYNACL?
--------------------------------------------------------------------------------

PyNaCl is a Python binding for libsodium, providing:
  - Symmetric encryption (SecretBox / XSalsa20-Poly1305)
  - Public-key encryption (Box / X25519-XSalsa20-Poly1305)
  - Digital signatures (Ed25519)
  - Hashing (SHA-256, SHA-512, BLAKE2b)

    import nacl
    print(nacl.__version__)    # 1.5.0

3) SECRETBOX (SYMMETRIC ENCRYPTION)
--------------------------------------------------------------------------------

SecretBox uses a single shared key for both encryption and decryption.
Uses XSalsa20 stream cipher + Poly1305 MAC.

    from nacl.secret import SecretBox
    from nacl.utils import random

    # Generate a random key
    key = random(SecretBox.KEY_SIZE)   # 32 bytes
    box = SecretBox(key)

    # Encrypt
    plaintext = b"confidential message"
    ciphertext = box.encrypt(plaintext)
    print(len(ciphertext))   # 32 + len(plaintext)  (nonce + tag added)

    # Decrypt
    decrypted = box.decrypt(ciphertext)
    print(decrypted)          # b'confidential message'

4) SECRETBOX WITH EXPLICIT NONCE
--------------------------------------------------------------------------------

    from nacl.secret import SecretBox
    from nacl.utils import nonce as make_nonce

    key = random(SecretBox.KEY_SIZE)
    box = SecretBox(key)

    # Generate a unique nonce (24 bytes)
    n = make_nonce(box)
    print(len(n))             # 24

    # Encrypt with explicit nonce
    ct = box.encrypt(b"message", n)

    # Decrypt (nonce is NOT included in ciphertext, must use same nonce)
    pt = box.decrypt(ct, n)
    print(pt)                 # b'message'

Note: Nonce must NEVER be reused with the same key. Use make_nonce() or
os.urandom(24) to generate unique nonces.

5) PUBLIC KEY BOX
--------------------------------------------------------------------------------

Box uses X25519 key exchange + XSalsa20-Poly1305 for each party.

    from nacl.public import PrivateKey, PublicKey, Box

    # Each party generates a key pair
    alice_sk = PrivateKey.generate()
    alice_pk = alice_sk.public_key

    bob_sk = PrivateKey.generate()
    bob_pk = bob_sk.public_key

    # Alice encrypts for Bob
    alice_box = Box(alice_sk, bob_pk)
    ciphertext = alice_box.encrypt(b"hello bob")

    # Bob decrypts from Alice
    bob_box = Box(bob_sk, alice_pk)
    plaintext = bob_box.decrypt(ciphertext)
    print(plaintext)          # b'hello bob'

    from nacl.signing import SigningKey

    # Public key encryption (X25519)
    sk = PrivateKey.generate()
    pk = sk.public_key
    print(bytes(sk).hex())    # 32 bytes, hex
    print(bytes(pk).hex())    # 32 bytes, hex

    # Signing (Ed25519)
    signing_key = SigningKey.generate()
    verify_key = signing_key.verify_key
    print(bytes(signing_key).hex())    # 32 bytes
    print(bytes(verify_key).hex())     # 32 bytes

7) DIGITAL SIGNATURES
--------------------------------------------------------------------------------

    from nacl.signing import SigningKey

    # Sign
    sk = SigningKey.generate()
    message = b"important document"
    signed = sk.sign(message)

    print(len(signed.signature))   # 64 bytes
    print(signed.message)          # original message

    # Verify
    vk = sk.verify_key
    verified = vk.verify(signed)
    print(verified)                # b'important document'

    # Bad signature
    sk2 = SigningKey.generate()
    try:
        vk2 = sk2.verify_key
        vk2.verify(signed)         # raises BadSignatureError
    except nacl.exceptions.BadSignatureError:
        print("bad signature detected!")

8) HASHING
--------------------------------------------------------------------------------

    from nacl.hash import sha256, sha512, blake2b

    # SHA-256 (returns hex-encoded bytes by default)
    h = sha256(b"hello world")
    print(h)                # 64-char hex string (bytes)

    # SHA-512
    h = sha512(b"hello world")
    print(h)                # 128-char hex string

    # BLAKE2b (with custom digest size)
    h = blake2b(b"hello world", digest_size=32)
    print(h)                # 64-char hex string

    # Raw binary (not hex-encoded)
    from nacl.encoding import RawEncoder
    h = sha256(b"hello world", encoder=RawEncoder)
    print(len(h))           # 32 bytes

    data = b"\xde\xad\xbe\xef"

    # Hex
    enc = HexEncoder.encode(data)
    print(enc)              # b'deadbeef'
    dec = HexEncoder.decode(enc)

    # Base64
    enc = Base64Encoder.encode(data)
    print(enc)              # b'3q2+7w=='
    dec = Base64Encoder.decode(enc)

    # Raw (no encoding)
    enc = RawEncoder.encode(data)
    print(enc)              # b'\xde\xad\xbe\xef'

10) COMMON PATTERNS
--------------------------------------------------------------------------------

Pattern: encrypt a password
    from nacl.secret import SecretBox
    from nacl.utils import random
    import hashlib

    password = b"my_password"
    key = hashlib.sha256(password).digest()
    box = SecretBox(key)
    ct = box.encrypt(b"secret data")

Pattern: key exchange (manual)
    from nacl.public import PrivateKey
    from nacl.bindings import crypto_scalarmult

    sk1 = PrivateKey.generate()
    sk2 = PrivateKey.generate()
    shared = crypto_scalarmult(bytes(sk1), bytes(sk2.public_key))
    # shared is a 32-byte shared secret

Pattern: signed JSON
    import json
    from nacl.signing import SigningKey

    sk = SigningKey.generate()
    data = json.dumps({"action": "transfer", "amount": 100}).encode()
    signed = sk.sign(data)

--------------------------------------------------------------------------------
11) PITFALLS & NOTES
--------------------------------------------------------------------------------

- Never reuse a nonce with the same SecretBox key
- SecretBox uses XSalsa20-Poly1305 (not AES-GCM)
- Box uses X25519-XSalsa20-Poly1305
- Ed25519 signatures are fixed 64 bytes
- private keys MUST be kept secret; never print or log them
- PyNaCl wraps libsodium, which handles constant-time comparisons internally
  to prevent timing attacks
- On Android, libsodium is statically linked into the .so via CFFI backend
- Wheels are tagged cp312-cp312-android_24
- CFFI backend is included (_cffi_backend.so) for libsodium bindings
- libsodium version is 1.0.20
