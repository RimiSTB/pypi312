--------------------------------------------------------------------------------
1) WHAT IS PSUTIL?
--------------------------------------------------------------------------------

psutil (process and system utilities) is a cross-platform library for accessing
system information in Python. It provides:
  - CPU usage, times, count, frequency
  - Memory usage (virtual, swap)
  - Disk usage, partitions, I/O counters
  - Network I/O, interfaces, connections
  - Process management (list, info, kill, suspend)
  - System boot time, uptime, users, sensors

    import psutil
    print(psutil.__version__)       # 6.1.1
    print(psutil.cpu_count())       # number of CPUs

3) CPU
--------------------------------------------------------------------------------

    import psutil

    # Number of logical CPUs
    print(psutil.cpu_count(logical=True))   # e.g. 8

    # Number of physical cores (may be None on some platforms)
    print(psutil.cpu_count(logical=False))  # e.g. 4

    # CPU times (user, system, idle, etc.)
    times = psutil.cpu_times()
    print("user:", times.user)
    print("system:", times.system)
    print("idle:", times.idle)

    # CPU percentage (blocks for interval seconds)
    print(psutil.cpu_percent(interval=0.1))   # e.g. 5.2

    # Per-CPU percentage
    print(psutil.cpu_percent(interval=0.1, percpu=True))  # [3.1, 1.2, ...]

    # CPU frequency (may return None on some platforms)
    freq = psutil.cpu_freq()
    if freq:
        print("current:", freq.current)

6) NETWORK
--------------------------------------------------------------------------------

    import psutil

    # I/O counters
    n = psutil.net_io_counters()
    print("bytes_sent:", n.bytes_sent)
    print("bytes_recv:", n.bytes_recv)
    print("packets_sent:", n.packets_sent)
    print("packets_recv:", n.packets_recv)

    # Per-interface I/O
    per = psutil.net_io_counters(pernic=True)
    for iface, counters in per.items():
        print(iface, counters.bytes_sent)

    # Interface addresses (IP + MAC)
    addrs = psutil.net_if_addrs()
    for iface, addr_list in addrs.items():
        for addr in addr_list:
            print(iface, addr.family, addr.address)

    # Interface stats (speed, MTU, isup)
    stats = psutil.net_if_stats()
    for iface, s in stats.items():
        print(iface, "up:", s.isup, "speed:", s.speed)

7) PROCESSES
--------------------------------------------------------------------------------

    import psutil
    import os

    # List all PIDs
    pids = psutil.pids()
    print("total processes:", len(pids))

    # Get current process
    p = psutil.Process(os.getpid())
    print("name:", p.name())
    print("status:", p.status())
    print("PID:", p.pid)
    print("PPID:", p.ppid())

    # Process memory
    m = p.memory_info()
    print("RSS:", m.rss)          # resident set size (bytes)
    print("VMS:", m.vms)          # virtual memory size

    # Process CPU
    cpu = p.cpu_times()
    print("user:", cpu.user)
    print("system:", cpu.system)

    # Process cmdline
    print("cmdline:", p.cmdline())

    # Process create time
    print("create_time:", p.create_time())

    # Iterating over all processes
    for proc in psutil.process_iter(["pid", "name", "status"]):
        if proc.info["status"] == psutil.STATUS_RUNNING:
            print("RUNNING:", proc.info["pid"], proc.info["name"])

9) COMMON MONITORING PATTERNS
--------------------------------------------------------------------------------

Pattern: wait for process
    import psutil, time
    p = psutil.Process(pid)
    while p.is_running():
        time.sleep(1)
    print("process %d terminated" % pid)

Pattern: memory usage of a process
    p = psutil.Process(pid)
    rss_mb = p.memory_info().rss / (1024 * 1024)
    print("RSS: %.1f MB" % rss_mb)

Pattern: system health check
    import psutil
    cpu = psutil.cpu_percent(interval=1)
    mem = psutil.virtual_memory()
    disk = psutil.disk_usage("/")
    print("CPU: %.1f%%, MEM: %.1f%%, DISK: %.1f%%" % (cpu, mem.percent, disk.percent))


--------------------------------------------------------------------------------
10) PITFALLS & NOTES
--------------------------------------------------------------------------------

- psutil.cpu_percent(interval=None) returns meaningless value on first call.
  Always use interval=0.1 or higher for meaningful results.
- On Android, some sensors (battery temperature, fan speed) may not be available.
- Process iteration can be slow if there are many processes.
- psutil.net_connections() may require root on some platforms.
- psutil.sensors_temperatures() returns empty dict on most Android devices.
- Wheels are tagged cp312-cp312-android_24. Install the correct ABI wheel.
- C extensions: _psutil_linux, _psutil_posix provide native functionality.
- No external dependencies required.
